import { cn } from '../lib/cn'

// Value / principle block: a short accent bar over a heading + body.
// Teal is the default signal; sage is the supporting second voice.
const bars = {
  teal: 'bg-teal-500',
  sage: 'bg-sage-500',
}

export default function ValueBlock({ title, body, tone = 'teal', className = '' }) {
  return (
    <div className={className}>
      <div className={cn('h-1 w-10 rounded-full', bars[tone])} />
      <h3 className="mt-4 text-lg font-semibold text-midnight-800">{title}</h3>
      <p className="mt-2 text-sm leading-relaxed text-midnight-600/90">{body}</p>
    </div>
  )
}
